# Asset ki profile generate karne wala function.
def profile_asset(asset_id):

    # Registry se saare registered assets retrieve karna.
    assets = get_all_assets()

    # Requested asset ko search karna.
    for asset in assets:

        # Database record se asset ki details nikalna.
        (
            registered_id,
            filename,
            modality,
            owner,
            permission_scope,
            processing_status,
            created_at
        ) = asset

        # Requested asset ID match karna.
        if registered_id == asset_id:

            # Filename se safe file name identify karna.
            safe_filename = Path(filename).name

            # Uploaded file ka actual path create karna.
            file_path = UPLOAD_DIR / safe_filename

            # Check karna ki uploaded file exist karti hai.
            file_exists = file_path.is_file()

            # File ka extension identify karna.
            file_extension = file_path.suffix.lower()

            # File ka size available hone par calculate karna.
            file_size = file_path.stat().st_size if file_exists else None

            # Modality supported hai ya nahi check karna.
            modality_supported = modality.lower() in SUPPORTED_MODALITIES

            # Asset ki complete profile prepare karna.
            profile = {
                "asset_id": registered_id,
                "filename": filename,
                "modality": modality,
                "asset_type": modality,
                "file_extension": file_extension,
                "file_exists": file_exists,
                "file_size_bytes": file_size,
                "modality_supported": modality_supported,
                "owner": owner,
                "permission_scope": permission_scope,
                "processing_status": processing_status,
                "created_at": created_at,
                "profile_status": (
                    "profiled"
                    if file_exists and modality_supported
                    else "needs_attention"
                )
            }

            # Generated asset profile return karna.
            return profile

    # Asset registry mein nahi mila toh None return karna.
    return None

 # Regular expression module import karna.
import re

# AI generated answer validate karne ka function.
def validate_answer(answer, evidence):

    # Validation issues store karne ke liye list.
    issues = []

    # Valid citations store karne ke liye list.
    valid_citations = []

    # Answer string check karna.
    if not isinstance(answer, str) or not answer.strip():

        # Empty answer par issue add karna.
        issues.append("Generated answer is empty.")

    # Evidence list check karna.
    if not evidence:

        # Evidence missing hone par issue add karna.
        issues.append("No supporting evidence available.")

    # Evidence ke har item ko check karna.
    for item in evidence or []:

        # Evidence metadata extract karna.
        metadata = item.get("metadata", {})

        # Important source identifiers select karna.
        for key in ["asset_id", "chunk_id", "segment_id", "source_id"]:

            # Metadata mein identifier available hai ya nahi.
            identifier = metadata.get(key)

            # Identifier answer mein present hai ya nahi.
            if identifier and re.search(
                r"(?<!\w)" + re.escape(str(identifier)) + r"(?!\w)",
                answer or ""
            ):

                # Supported citation record karna.
                valid_citations.append(str(identifier))

    # Evidence available hai lekin citation mention nahi hui.
    if evidence and not valid_citations:

        # Missing citation warning add karna.
        issues.append("No retrieved evidence identifiers found in answer.")

    # Duplicate citations remove karna.
    valid_citations = list(dict.fromkeys(valid_citations))

    # Final validation result return karna.
    return {
        "is_valid": len(issues) == 0,
        "issues": issues,
        "valid_citations": valid_citations
    }

# Regular expressions use karne ke liye re import karna.
import re

# Existing query processor import karna.
from retrieval.query_processor import process_query


# Project ki supported modalities define karna.
SUPPORTED_MODALITIES = {
    "pdf": ["pdf", "document", "notes"],
    "video": ["video", "lecture"],
    "audio": ["audio", "recording", "speech"],
    "image": ["image", "picture", "diagram", "chart"],
    "text": ["text", "article"],
    "quiz": ["quiz", "question", "assessment"],
    "discussion": ["discussion", "forum"]
}


# Query se retrieval plan generate karne wala function.
def plan_retrieval(query, k=3):

    # User query ko clean karna.
    clean_query = process_query(query)

    # Top K value positive integer honi chahiye.
    if isinstance(k, bool) or not isinstance(k, int) or k <= 0:
        raise ValueError("k must be a positive integer")

    # Query ko lowercase mein convert karna.
    lower_query = clean_query.lower()

    # Query mein mentioned modalities identify karna.
    selected_modalities = []

    # Har supported modality ko check karna.
    for modality, terms in SUPPORTED_MODALITIES.items():

        # Modality ke keywords ko query mein search karna.
        if any(
            re.search(r"\b" + re.escape(term) + r"\b", lower_query)
            for term in terms
        ):

            # Matching modality ko result mein add karna.
            selected_modalities.append(modality)

    # Common unnecessary words define karna.
    stop_words = {
        "the", "is", "are", "was", "were", "a", "an",
        "and", "or", "of", "in", "on", "to", "for",
        "from", "with", "by", "using", "explain",
        "describe", "show", "me", "please", "what",
        "how", "why", "about"
    }

    # Query ko individual words mein divide karna.
    words = re.findall(r"\b[a-zA-Z0-9]+\b", lower_query)

    # Unnecessary words remove karke keywords prepare karna.
    keywords = [
        word for word in words
        if word not in stop_words
        and not any(
            word in terms
            for terms in SUPPORTED_MODALITIES.values()
        )
    ]

    # Agar keywords empty hain toh original query use karna.
    if not keywords:
        keywords = words

    # Final structured retrieval plan create karna.
    retrieval_plan = {
        "original_query": clean_query,
        "keywords": list(dict.fromkeys(keywords)),
        "modality_filters": selected_modalities,
        "top_k": k,
        "strategy": (
            "modality_filtered"
            if selected_modalities
            else "search_all_modalities"
        )
    }

    # Prepared retrieval plan return karna.
    return retrieval_plan

# Existing RAG answer generator import karna.
from rag.answer_generator import generate_answer

# Existing quality validator import karna.
from agents.quality_validator import validate_answer


# Evidence ke basis par final answer generate karna.
def synthesize_answer(question, evidence, llm):

    # Existing RAG generator se answer generate karna.
    generated_result = generate_answer(
        question,
        evidence,
        llm
    )

    # Check karna ki generator ne dictionary return ki hai.
    if isinstance(generated_result, dict):

        # Dictionary se generated answer extract karna.
        answer = generated_result.get("answer", "")

        # Generator se available evidence gaps extract karna.
        evidence_gaps = generated_result.get("evidence_gaps")

    else:

        # Direct AI response ko answer variable mein store karna.
        answer = generated_result

        # Evidence gaps available nahi hain.
        evidence_gaps = None

    # Existing validator se answer ki quality check karna.
    validation = validate_answer(
        answer,
        evidence
    )

    # Final structured synthesis result create karna.
    result = {
        "question": question,
        "answer": answer,
        "evidence_count": len(evidence or []),
        "validation": validation,
        "evidence_gaps": evidence_gaps,
        "status": (
            "validated"
            if validation["is_valid"]
            else "needs_review"
        )
    }

    # Final result return karna.
    return result

from pathlib import Path


# Supported file formats
SUPPORTED_FORMATS = {
    ".mp4": "video",
    ".mov": "video",
    ".mp3": "audio",
    ".wav": "audio",
    ".pdf": "document",
    ".png": "image",
    ".jpg": "image",
    ".jpeg": "image",
    ".txt": "text",
    ".csv": "quiz"
}


# Maximum file size: 200 MB
MAX_FILE_SIZE = 200 * 1024 * 1024


def validate_file(file_path):

    file_path = Path(file_path)

    # Check whether file exists
    if not file_path.exists():
        return False, "File does not exist", None

    # Check whether it is a file
    if not file_path.is_file():
        return False, "Path is not a file", None

    # Check file extension
    extension = file_path.suffix.lower()

    if extension not in SUPPORTED_FORMATS:
        return False, "Unsupported file format", None

    # Check empty file
    if file_path.stat().st_size == 0:
        return False, "File is empty", None

    # Check maximum size
    if file_path.stat().st_size > MAX_FILE_SIZE:
        return False, "File exceeds 200 MB limit", None

    modality = SUPPORTED_FORMATS[extension]

    return True, "File validation successful", modality

from uuid import uuid4


def create_text_chunks(
    pages,
    asset_id,
    chunk_size=500,
    overlap=50
):

    chunks = []

    for page in pages:

        text = page["text"]

        start = 0

        while start < len(text):

            end = start + chunk_size

            chunk_text = text[start:end].strip()

            if chunk_text:

                chunks.append({

                    "chunk_id": str(uuid4()),

                    "asset_id": asset_id,

                    "source_file": page["source_file"],

                    "segment_id": page.get("segment_id"),

                    "page_number": page.get("page_number"),

                    "modality": page.get("modality", "document"),

                    "chunk_text": chunk_text

                })

            start = end - overlap

    return chunks

import re  # Text cleaning ke liye.

def process_text(text_path, asset_id):  # Text preprocessing function define kar rahe hain.

    text_path = Path(text_path)  # File path ko Path object mein convert kar rahe hain.

    if not text_path.exists():  # Check kar rahe hain ki text file available hai ya nahi.

    segments = []  # Clean text segments store karne ke liye list bana rahe hain.

    seen = set()  # Duplicate sentences track karne ke liye set bana rahe hain.

    for sentence in paragraphs:  # Har sentence ko process kar rahe hain.

        if sentence and sentence.lower() not in seen:  # Empty aur duplicate sentences skip kar rahe hain.

            seen.add(sentence.lower())  # Sentence ko duplicate tracking mein add kar rahe hain.

            segments.append({  # Structured text segment bana rahe hain.

                "asset_id": asset_id,  # Original asset ki unique ID.

                "source_file": text_path.name,  # Original text filename.

                "modality": "text",  # Asset ki modality define kar rahe hain.

                "segment_id": f"{asset_id}-SEG-{len(segments) + 1}",  # Unique segment ID bana rahe hain.

                "text": sentence  # Clean sentence save kar rahe hain.

            })

    return segments  # Saare clean segments return kar rahe hain.

def extract_video_frames(video_path, asset_id, interval_seconds=5):
    # Ye function video se fixed interval par frames extract karega.

    video_path = Path(video_path)
    # Video file ke path ko Path format mein convert kar rahe hain.

    if not video_path.exists():
        # Check kar rahe hain ki video file exist karti hai ya nahi.

        raise FileNotFoundError("Video file not found")
        # File nahi mili to error show hoga.

    if video_path.suffix.lower() not in [".mp4", ".mov"]:
        # Check kar rahe hain ki video MP4 ya MOV format mein hai.

        raise ValueError("Only MP4 and MOV videos are supported")
        # Doosra format hua to error show hoga.

    if interval_seconds <= 0:
        # Check kar rahe hain ki frame interval zero ya negative na ho.

        raise ValueError("Interval must be greater than zero")
        # Galat interval hone par error show hoga.

    project_dir = Path(__file__).resolve().parent.parent
    # Project ka main folder locate kar rahe hain.

    output_dir = project_dir / "data" / "processed" / asset_id / "frames"
    # Is video ke frames save karne ke liye folder ka path bana rahe hain.

    output_dir.mkdir(parents=True, exist_ok=True)
    # Folder exist nahi karta to automatically create hoga.

    cap = cv2.VideoCapture(str(video_path))
    # OpenCV ke through video open kar rahe hain.

    if not cap.isOpened():
        # Check kar rahe hain ki video successfully open hui ya nahi.

        raise ValueError("Unable to open video")
        # Video open nahi hui to error show hoga.

    fps = cap.get(cv2.CAP_PROP_FPS)
    # Video ke frames per second (FPS) read kar rahe hain.

    if fps <= 0:
        # Check kar rahe hain ki FPS valid hai ya nahi.

        cap.release()
        # Video resource release kar rahe hain.

        raise ValueError("Invalid video FPS")
        # FPS invalid hua to error show hoga.

    frame_step = max(1, round(fps * interval_seconds))
    # Calculate kar rahe hain ki kitne frames skip karne hain.

    extracted_frames = []
    # Extracted frames ki information is list mein store hogi.

    frame_index = 0
    # Video ka frame counter zero se start hoga.

    try:
        # Video processing ko safely run karne ke liye try block.

        while True:
            # Jab tak video mein frames milte rahenge, loop chalega.

            success, frame = cap.read()
            # Video se next frame read kar rahe hain.

            if not success:
                # Agar frame nahi mila to video khatam ho gayi.

                break
                # Loop stop kar rahe hain.

            if frame_index % frame_step == 0:
                # Check kar rahe hain ki ye frame extract karna hai ya nahi.

                timestamp = frame_index / fps
                # Frame ka timestamp seconds mein calculate kar rahe hain.

                frame_id = str(uuid4())
                # Frame ko unique ID de rahe hain.

                frame_name = f"frame_{frame_index:06d}.jpg"
                # Frame ka filename create kar rahe hain.

                frame_path = output_dir / frame_name
                # Frame save karne ki location bana rahe hain.

                saved = cv2.imwrite(str(frame_path), frame)
                # Extracted image ko JPG file mein save kar rahe hain.

                if not saved:
                    # Check kar rahe hain ki image successfully save hui ya nahi.

                    raise ValueError("Unable to save extracted frame")
                    # Save failure hone par error show hoga.

                    # Frame ki details list mein add kar rahe hain.

                    "frame_id": frame_id,
                    # Har frame ki unique identity.

                    "asset_id": asset_id,
                    # Original video ki asset identity.

                    "source_file": video_path.name,

                    # Video mein frame kis time par tha.

                    "frame_path": str(frame_path),
                    # Extracted image ki file location.

                    "modality": "video_frame"
                    # Batata hai ki ye image video se extract hui hai.
                })
                # Frame ki saari metadata information complete.

            frame_index += 1

    finally:
        # Processing complete ho ya error aaye, ye block chalega.

        cap.release()
        # Video resource release kar rahe hain.

# FAISS vector database create karna.
def create_vector_store(chunks):

    # Chunk text ko documents ke format mein convert karna.
    texts = [chunk["chunk_text"] for chunk in chunks]

    # Chunk metadata preserve karna.
    metadatas = [
        {
            key: value
            for key, value in chunk.items()
            if key not in ["chunk_text"]
        }
        for chunk in chunks
    ]

    # LangChain FAISS vector store create karna.
    vector_store = FAISS.from_texts(
        texts=texts,
        embedding=embeddings_model,
        metadatas=metadatas
    )

    # Created vector store return karna.
    return vector_store

def save_vector_store(vector_store, path="data/vector_store"):

    # FAISS database ko local storage mein save karna.
    vector_store.save_local(path)

    print("Vector store saved successfully!")

def load_vector_store(path="data/vector_store"):

    # Local FAISS database load karna.
    vector_store = FAISS.load_local(
        path,
        embeddings_model,
        allow_dangerous_deserialization=True
    )

    # Loaded vector store return karna.
    return vector_store


# Similarity search with optional modality filtering.
def search_vector_store(
    vector_store,
    query,
    k=3,
    modality_filters=None
):

    # Default mein metadata filter disabled rakhna.
    metadata_filter = None

    # Check karna ki modality filters available hain.
    if modality_filters:

        # Requested modalities ke according metadata filter banana.
        metadata_filter = {
            "modality": {
                "$in": modality_filters
            }
        }

    # FAISS se relevant documents aur distance scores retrieve karna.
    results = vector_store.similarity_search_with_score(
        query,
        k=k,
        filter=metadata_filter,
        fetch_k=max(k * 10, 20)
    )

    # Retrieved results return karna.
    return results

 # Regular expression module import karna.
import re

# AI generated answer validate karne ka function.
def validate_answer(answer, evidence):

    # Validation issues store karne ke liye list.
    issues = []

    # Valid citations store karne ke liye list.
    valid_citations = []

    # Answer string check karna.
    if not isinstance(answer, str) or not answer.strip():

        # Empty answer par issue add karna.
        issues.append("Generated answer is empty.")

    # Evidence list check karna.
    if not evidence:

        # Evidence missing hone par issue add karna.
        issues.append("No supporting evidence available.")

    # Evidence ke har item ko check karna.
    for item in evidence or []:

        # Evidence metadata extract karna.
        metadata = item.get("metadata", {})

        # Important source identifiers select karna.
        for key in ["asset_id", "chunk_id", "segment_id", "source_id"]:

            # Metadata mein identifier available hai ya nahi.
            identifier = metadata.get(key)

            # Identifier answer mein present hai ya nahi.
            if identifier and re.search(
                r"(?<!\w)" + re.escape(str(identifier)) + r"(?!\w)",
                answer or ""
            ):

                # Supported citation record karna.
                valid_citations.append(str(identifier))

    # Evidence available hai lekin citation mention nahi hui.
    if evidence and not valid_citations:

        # Missing citation warning add karna.
        issues.append("No retrieved evidence identifiers found in answer.")

    # Duplicate citations remove karna.
    valid_citations = list(dict.fromkeys(valid_citations))

    # Final validation result return karna.
    return {
        "is_valid": len(issues) == 0,
        "issues": issues,
        "valid_citations": valid_citations
    }

# Regular expressions use karne ke liye re import karna.
import re

# Existing query processor import karna.
from retrieval.query_processor import process_query


# Project ki supported modalities define karna.
SUPPORTED_MODALITIES = {
    "pdf": ["pdf", "document", "notes"],
    "video": ["video", "lecture"],
    "audio": ["audio", "recording", "speech"],
    "image": ["image", "picture", "diagram", "chart"],
    "text": ["text", "article"],
    "quiz": ["quiz", "question", "assessment"],
    "discussion": ["discussion", "forum"]
}


# Query se retrieval plan generate karne wala function.
def plan_retrieval(query, k=3):

    # User query ko clean karna.
    clean_query = process_query(query)

    # Top K value positive integer honi chahiye.
    if isinstance(k, bool) or not isinstance(k, int) or k <= 0:
        raise ValueError("k must be a positive integer")

    # Query ko lowercase mein convert karna.
    lower_query = clean_query.lower()

    # Query mein mentioned modalities identify karna.
    selected_modalities = []

    # Har supported modality ko check karna.
    for modality, terms in SUPPORTED_MODALITIES.items():

        # Modality ke keywords ko query mein search karna.
        if any(
            re.search(r"\b" + re.escape(term) + r"\b", lower_query)
            for term in terms
        ):

            # Matching modality ko result mein add karna.
            selected_modalities.append(modality)

    # Common unnecessary words define karna.
    stop_words = {
        "the", "is", "are", "was", "were", "a", "an",
        "and", "or", "of", "in", "on", "to", "for",
        "from", "with", "by", "using", "explain",
        "describe", "show", "me", "please", "what",
        "how", "why", "about"
    }

    # Query ko individual words mein divide karna.
    words = re.findall(r"\b[a-zA-Z0-9]+\b", lower_query)

    # Unnecessary words remove karke keywords prepare karna.
    keywords = [
        word for word in words
        if word not in stop_words
        and not any(
            word in terms
            for terms in SUPPORTED_MODALITIES.values()
        )
    ]

    # Agar keywords empty hain toh original query use karna.
    if not keywords:
        keywords = words

    # Final structured retrieval plan create karna.
    retrieval_plan = {
        "original_query": clean_query,
        "keywords": list(dict.fromkeys(keywords)),
        "modality_filters": selected_modalities,
        "top_k": k,
        "strategy": (
            "modality_filtered"
            if selected_modalities
            else "search_all_modalities"
        )
    }

    # Prepared retrieval plan return karna.
    return retrieval_plan

from pathlib import Path


# Supported file formats
SUPPORTED_FORMATS = {
    ".mp4": "video",
    ".mov": "video",
    ".mp3": "audio",
    ".wav": "audio",
    ".pdf": "document",
    ".png": "image",
    ".jpg": "image",
    ".jpeg": "image",
    ".txt": "text",
    ".csv": "quiz"
}


# Maximum file size: 200 MB
MAX_FILE_SIZE = 200 * 1024 * 1024


def validate_file(file_path):

    file_path = Path(file_path)

    # Check whether file exists
    if not file_path.exists():
        return False, "File does not exist", None

    # Check whether it is a file
    if not file_path.is_file():
        return False, "Path is not a file", None

    # Check file extension
    extension = file_path.suffix.lower()

    if extension not in SUPPORTED_FORMATS:
        return False, "Unsupported file format", None

    # Check empty file
    if file_path.stat().st_size == 0:
        return False, "File is empty", None

    # Check maximum size
    if file_path.stat().st_size > MAX_FILE_SIZE:
        return False, "File exceeds 200 MB limit", None

    modality = SUPPORTED_FORMATS[extension]

    return True, "File validation successful", modality

def extract_pdf_pages(pdf_path, asset_id):  # PDF processing function.

    pdf_path = Path(pdf_path)  # PDF path convert kar rahe hain.

    if not pdf_path.exists():  # PDF file check kar rahe hain.
        raise FileNotFoundError(f"PDF not found: {pdf_path}")  # Missing file par error.

    pages_data = []  # Processed pages store karne ke liye list.

    with fitz.open(pdf_path) as pdf:  # PDF open kar rahe hain.

        for page_number, page in enumerate(pdf, start=1):  # Har page read kar rahe hain.

            text = page.get_text("text").strip()  # Normal PDF text extract kar rahe hain.

            if not text:  # Agar page mein selectable text nahi mila.

                pix = page.get_pixmap(dpi=200)  # PDF page ko image mein convert kar rahe hain.

                text = pytesseract.image_to_string(image).strip()  # OCR se scanned page ka text nikal rahe hain.

                extraction_method = "tesseract_ocr"  # OCR method record kar rahe hain.

            page_data = {  # Page ka structured metadata bana rahe hain.

                "page_number": page_number,  # Original page number.

                "text": text,  # Extracted page text.

                "has_text": bool(text),  # Text available hai ya nahi.

                "extraction_method": extraction_method  # Text kis method se nikla.

            }

            pages_data.append(page_data)  # Page data list mein add kar rahe hain.

from uuid import uuid4


def create_text_chunks(
    pages,
    asset_id,
    chunk_size=500,
    overlap=50
):

    chunks = []

    for page in pages:

        text = page["text"]

        start = 0

        while start < len(text):

            end = start + chunk_size

            chunk_text = text[start:end].strip()

            if chunk_text:

                chunks.append({

                    "chunk_id": str(uuid4()),

                    "asset_id": asset_id,

                    "source_file": page["source_file"],

                    "segment_id": page.get("segment_id"),

                    "page_number": page.get("page_number"),

                    "modality": page.get("modality", "document"),

                    "chunk_text": chunk_text

                })

            start = end - overlap

    return chunks

        if sentence and sentence.lower() not in seen:  # Empty aur duplicate sentences skip kar rahe hain.

            seen.add(sentence.lower())  # Sentence ko duplicate tracking mein add kar rahe hain.

            segments.append({  # Structured text segment bana rahe hain.

                "asset_id": asset_id,  # Original asset ki unique ID.

                "source_file": text_path.name,  # Original text filename.

                "modality": "text",  # Asset ki modality define kar rahe hain.

                "segment_id": f"{asset_id}-SEG-{len(segments) + 1}",  # Unique segment ID bana rahe hain.

import cv2
# OpenCV library video ko read aur frames extract karne ke liye hai.

from pathlib import Path
# Path folders aur file locations handle karne ke liye hai.

from uuid import uuid4
# Har extracted frame ko unique ID dene ke liye hai.


def extract_video_frames(video_path, asset_id, interval_seconds=5):
    # Ye function video se fixed interval par frames extract karega.

    video_path = Path(video_path)
    # Video file ke path ko Path format mein convert kar rahe hain.

    if not video_path.exists():
        # Check kar rahe hain ki video file exist karti hai ya nahi.

        raise FileNotFoundError("Video file not found")
        # File nahi mili to error show hoga.

    if video_path.suffix.lower() not in [".mp4", ".mov"]:
        # Check kar rahe hain ki video MP4 ya MOV format mein hai.

        raise ValueError("Only MP4 and MOV videos are supported")
        # Doosra format hua to error show hoga.

    if interval_seconds <= 0:
        # Check kar rahe hain ki frame interval zero ya negative na ho.

        raise ValueError("Interval must be greater than zero")
        # Galat interval hone par error show hoga.

    project_dir = Path(__file__).resolve().parent.parent
    # Project ka main folder locate kar rahe hain.

    output_dir = project_dir / "data" / "processed" / asset_id / "frames"
    # Is video ke frames save karne ke liye folder ka path bana rahe hain.

    output_dir.mkdir(parents=True, exist_ok=True)
    # Folder exist nahi karta to automatically create hoga.

    cap = cv2.VideoCapture(str(video_path))
    # OpenCV ke through video open kar rahe hain.

    if not cap.isOpened():
        # Check kar rahe hain ki video successfully open hui ya nahi.

        raise ValueError("Unable to open video")
        # Video open nahi hui to error show hoga.

    fps = cap.get(cv2.CAP_PROP_FPS)
    # Video ke frames per second (FPS) read kar rahe hain.

    if fps <= 0:
        # Check kar rahe hain ki FPS valid hai ya nahi.

        cap.release()
        # Video resource release kar rahe hain.

        raise ValueError("Invalid video FPS")
        # FPS invalid hua to error show hoga.

    frame_step = max(1, round(fps * interval_seconds))
    # Calculate kar rahe hain ki kitne frames skip karne hain.

    extracted_frames = []
    # Extracted frames ki information is list mein store hogi.

    frame_index = 0
    # Video ka frame counter zero se start hoga.

    try:
        # Video processing ko safely run karne ke liye try block.

        while True:
            # Jab tak video mein frames milte rahenge, loop chalega.

            success, frame = cap.read()
            # Video se next frame read kar rahe hain.

            if not success:
                # Agar frame nahi mila to video khatam ho gayi.

                break
                # Loop stop kar rahe hain.

            if frame_index % frame_step == 0:
                # Check kar rahe hain ki ye frame extract karna hai ya nahi.

                timestamp = frame_index / fps
                # Frame ka timestamp seconds mein calculate kar rahe hain.

                frame_id = str(uuid4())
                # Frame ko unique ID de rahe hain.

                frame_name = f"frame_{frame_index:06d}.jpg"
                # Frame ka filename create kar rahe hain.

                frame_path = output_dir / frame_name
                # Frame save karne ki location bana rahe hain.

                saved = cv2.imwrite(str(frame_path), frame)
                # Extracted image ko JPG file mein save kar rahe hain.

                if not saved:
                    # Check kar rahe hain ki image successfully save hui ya nahi.

                    raise ValueError("Unable to save extracted frame")
                    # Save failure hone par error show hoga.

                extracted_frames.append({
                    # Frame ki details list mein add kar rahe hain.

                    "frame_id": frame_id,
                    # Har frame ki unique identity.

                    "asset_id": asset_id,
                    # Original video ki asset identity.

                    "source_file": video_path.name,
                    # Original video ka filename.

                    "timestamp_seconds": round(timestamp, 2),
                    # Video mein frame kis time par tha.

                    "frame_path": str(frame_path),
                    # Extracted image ki file location.

                    "modality": "video_frame"
                    # Batata hai ki ye image video se extract hui hai.
                })
                # Frame ki saari metadata information complete.

            frame_index += 1
            # Agle frame ke liye counter increase kar rahe hain.

    finally:
        # Processing complete ho ya error aaye, ye block chalega.

        cap.release()
        # Video resource release kar rahe hain.

    return extracted_frames
    # Extracted frames ki complete information return kar rahe hain.

def create_vector_store(chunks):

    # Chunk text ko documents ke format mein convert karna.
    texts = [chunk["chunk_text"] for chunk in chunks]

    # Chunk metadata preserve karna.
    metadatas = [
        {
            key: value
            for key, value in chunk.items()
            if key not in ["chunk_text"]
        }
        for chunk in chunks
    ]

    # LangChain FAISS vector store create karna.
    vector_store = FAISS.from_texts(
        texts=texts,
        embedding=embeddings_model,
        metadatas=metadatas
    )

    # Created vector store return karna.
    return vector_store
